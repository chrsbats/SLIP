@pytest.mark.asyncio
async def test_bound_expression_is_source_order_functions_and_live_getters():
    runner = ScriptRunner()
    result = await runner.handle_script("calculate: fn {x, y} [x + y * 2]")
    assert result.status == "ok", result.error_message
    function = runner.root_scope["calculate"]
    method = function.methods[0]
    terms = runner.evaluator._program(method.body[0], method)

    assert isinstance(terms, list) and len(terms) == 5
    scope = Scope(bindings={"x": 4, "y": 5})
    assert callable(terms[0]) and terms[0](scope) == 4
    assert terms[1] is runner.root_scope["+"]
    assert callable(terms[2]) and terms[2](scope) == 5
    assert terms[3] is runner.root_scope["*"]
    assert terms[4] == 2
    for x, y in ((10, 5), (3, 4)):
        assert await runner.evaluator.call(
            function, [x, y], runner.root_scope,
        ) == (x + y) * 2
        assert runner.evaluator._program(method.body[0], method) is terms

@pytest.mark.asyncio
async def test_captured_python_calls_reuse_adapters_and_keep_value_identity(
    monkeypatch,
):
    from slip.slip_datatypes import NativeFunction

    def increment(value):
        return value + 1

    runner = ScriptRunner()
    runner.root_scope["host-increment"] = increment
    runner.root_scope["same-function"] = lambda value: value is increment
    loaded = await runner.handle_script("""
invoke-host: fn {value} [host-increment value]
inspect-host: fn {} [same-function host-increment]
""")
    assert loaded.status == "ok", loaded.error_message

    def unexpected(*args, **kwargs):
        raise AssertionError("a captured native callable rebuilt its adapter")

    monkeypatch.setattr(NativeFunction, "__init__", unexpected)
    for value in (1, 10):
        assert await runner.evaluator.call(
            runner.root_scope["invoke-host"], [value], runner.root_scope,
        ) == value + 1
        assert await runner.evaluator.call(
            runner.root_scope["inspect-host"], [], runner.root_scope,
        ) is True

@pytest.mark.asyncio
async def test_dispatch_resolves_prototype_once_per_candidate(monkeypatch):
    runner = ScriptRunner()
    loaded = await runner.handle_script("""
Player: scope #{}
describe: fn {x: Player} [1]
describe: fn {x} [0]
""")
    assert loaded.status == "ok", loaded.error_message
    function = runner.root_scope["describe"]
    annotation = function.methods[0].signature._parameters[0].annotation
    reads = []
    original = runner.evaluator.path_resolver.get

    async def observe(path, scope):
        if path is annotation:
            reads.append(path)
        return await original(path, scope)

    monkeypatch.setattr(runner.evaluator.path_resolver, "get", observe)
    assert await runner.evaluator.call(
        function, [Scope(parent=runner.root_scope["Player"])],
        runner.root_scope,
    ) == 1
    assert len(reads) == 1

@pytest.mark.asyncio
@pytest.mark.parametrize('append', [False, True])
async def test_repeated_template_edits_keep_the_original_source(append):
    from slip.slip_datatypes import Group

    runner = ScriptRunner()
    calls = []

    def source():
        return runner.root_scope['probe'].methods[0].body[0][0]

    def first():
        calls.append('first')
        source().nodes[2][0] = Group([[3]])
        return 0

    def second():
        calls.append('second')
        source().nodes[2] = [Group([[9]])]
        if append:
            source().nodes.append([Group([[11]])])
        return 0

    runner.root_scope['first-edit'] = first
    runner.root_scope['second-edit'] = second
    result = await runner.handle_script(
        'probe: fn {} [[(inject (first-edit)); '
        '(inject (second-edit)); (2)]]; run probe'
    )
    assert result.status == 'ok', result.error_message
    assert result.value == (11 if append else 9)
    assert calls == ['first', 'second']

@pytest.mark.asyncio
@pytest.mark.parametrize("name", [
    "loop-sum", "function-calls", "traverse-objects",
])
async def test_production_workloads_execute_prepared_call_lists(
    name, monkeypatch,
):
    from slip import slip_prepare

    runner, _ = await python_overhead.prepare()
    _, args, expected = python_overhead.make_cases()[name]
    original = copy.deepcopy(args)

    def source_interpretation(*args, **kwargs):
        raise AssertionError("prepared execution reentered source reduction")

    for method in (
        "_interpret_expr", "_eval_prefix", "_reduce_tail", "_read_term",
        "_prepared_operand",
    ):
        monkeypatch.setattr(
            runner.evaluator, method, source_interpretation, raising=False,
        )
    monkeypatch.setattr(slip_prepare, "operand", source_interpretation)
    for _ in range(2):
        assert await runner.evaluator.call(
            runner.root_scope[name], args, runner.root_scope,
        ) == expected
    assert args == original
    assert runner.evaluator.side_effects == []
    assert runner.evaluator.call_stack == []
