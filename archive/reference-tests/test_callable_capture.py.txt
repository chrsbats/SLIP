@pytest.mark.asyncio
async def test_prepared_operands_read_current_frames_and_replacement_objects():
    runner = ScriptRunner()
    result = await runner.handle_script("""
calculate: fn {obj, extra} [obj.stats.weight + extra]
""")
    assert result.status == "ok", result.error_message
    function = runner.root_scope["calculate"]
    obj = {"stats": {"weight": 4}}
    for weight, extra in ((4, 1), (8, 2)):
        obj["stats"]["weight"] = weight
        assert await runner.evaluator.call(
            function, [obj, extra], runner.root_scope,
        ) == weight + extra
    obj = {"stats": {"weight": 20}}
    assert await runner.evaluator.call(
        function, [obj, 3], runner.root_scope,
    ) == 23


@pytest.mark.asyncio
async def test_preparation_and_foreach_lookahead_do_not_read_host_properties():
    class Object:
        reads = 0

        @property
        def visible(self):
            self.reads += 1
            return True

    obj = Object()
    runner = ScriptRunner()
    runner.root_scope["obj"] = obj
    result = await runner.handle_script("""
observe: fn {} [
    foreach {i} #[1, 2, 3] [if [obj.visible] [emit 'visible' i]]
]
""")
    assert result.status == "ok", result.error_message
    assert obj.reads == 0
    result = await runner.handle_script("observe")
    assert result.status == "ok", result.error_message
    assert obj.reads == 3
    assert [effect["message"] for effect in result.side_effects] == [1, 2, 3]

@pytest.mark.asyncio
async def test_foreach_live_member_operator_accumulates_with_one_read():
    class Operators:
        reads = 0

        @property
        def plus(self):
            self.reads += 1
            return PipedCallable(lambda a, b: a + b, "add")

    operators = Operators()
    runner = ScriptRunner()
    runner.root_scope["ops"] = operators
    result = await runner.handle_script("""
total: 0
foreach {i} #[1, 2, 3] [total ops.plus i]
total
""")
    assert result.status == "ok", result.error_message
    assert result.value == 6
    assert operators.reads == 3

@pytest.mark.asyncio
async def test_prepared_function_body_remains_editable_source_code():
    runner = ScriptRunner()
    result = await runner.handle_script("calculate: fn {x} [x + 1]")
    assert result.status == "ok", result.error_message
    function = runner.root_scope["calculate"]
    method = function.methods[0]
    assert await runner.evaluator.call(
        function, [10], runner.root_scope,
    ) == 11
    method.body[0][1].segments[0].text = "-"
    assert await runner.evaluator.call(
        function, [10], runner.root_scope,
    ) == 9
    method.body[0] = [GetPath([Name("x")]), GetPath([Name("*")]), 3]
    assert await runner.evaluator.call(
        function, [10], runner.root_scope,
    ) == 30

        "value": 2,
    }


@pytest.mark.asyncio
async def test_known_calls_and_operands_skip_symbolic_resolution(monkeypatch):
    runner = ScriptRunner()
    result = await runner.handle_script("calculate: fn {a, b} [a + b]")
    assert result.status == "ok", result.error_message

    def unexpected(*args):
        raise AssertionError("prepared expression entered symbolic resolution")

    monkeypatch.setattr(runner.evaluator.path_resolver, "read", unexpected)
    for a, b in ((2, 3), (20, 30)):
        assert await runner.evaluator.call(
            runner.root_scope["calculate"], [a, b], runner.root_scope,
        ) == a + b


@pytest.mark.asyncio
async def test_cached_module_captures_use_each_runners_live_host_data(
    tmp_path, monkeypatch,
):
    from slip import slip_file

    monkeypatch.setenv("SLIP_CACHE_DIR", str(tmp_path / "cache"))
    monkeypatch.setattr(slip_file, "_compiled_slip_cache", {})
    module = tmp_path / "commands.slip"
    module.write_text(
        "lookup: fn {object-id: `string`} ["
        "obj: host-object object-id; obj.value] |public\n"
    )
    commands = []
    worlds = []
    for label in ("first", "second"):
        world = {"id:item": {"value": label}}
        worlds.append(world)
        runner = ScriptRunner(host_data=world.__getitem__)
        commands.append(await runner.import_public_module(
            f"file://{module.as_posix()}"
        ))
        slip_file._compiled_slip_cache.clear()

    payload = {"function": "lookup", "arguments": {"object-id": "id:item"}}
    for index, label in ((0, "first"), (1, "second"), (0, "replacement")):
        if label == "replacement":
            worlds[index]["id:item"] = {"value": label}
        result = await commands[index].run_json_command(payload)
        assert result.status == "ok", result.error_message
        assert result.value == label

@pytest.mark.asyncio
async def test_example_method_guards_keep_forward_names_dynamic():
    runner = ScriptRunner()
    result = await runner.handle_script('''
choose: fn {x} [x] |where [check x] |example {x: 1 -> 1}
check: fn {x} [true]
''')
    assert result.status == 'ok', result.error_message
    function = runner.root_scope['choose']
    assert await runner.evaluator.call(function, [7], runner.root_scope) == 7
    runner.root_scope['check'] = runner.root_scope['core-not']
    with pytest.raises(TypeError, match='No matching method'):
        await runner.evaluator.call(function, [7], runner.root_scope)
